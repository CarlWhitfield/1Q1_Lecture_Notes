\textbf{Example 3.4.1}

\begin{align*}
    \int \frac{(\ln x)^5}{x} \, {\rm d}x
\end{align*}
Substitute $u = \ln x$, so ${\rm d} u/{\rm d}x = 1/x$, or $(1/x){\rm d}x = {\rm d} u$
\begin{align*}
    \int u^5 \, {\rm d}u = \frac{u^6}{6} +C = \frac{(\ln x)^6}{6} +C.
\end{align*}

\textbf{Example 3.4.2}

\begin{align*}
    \int_0^{1/2} x \cos(\pi x^2) \, {\rm d}x
\end{align*}
Substitute $u = \pi x^2$ so ${\rm d} u/{\rm d}x = 2 \pi x $, or $x{\rm d}x = {\rm d} u / (2\pi)$. The limits become $u=0$ and $u = \pi/4$
\begin{align*}
\frac{1}{2 \pi} \int_0^{\pi/4} \cos u \, {\rm d}u = \frac{1}{2 \pi} [\sin u]_0^{\pi/4} = \frac{\sqrt{2}}{4\pi}
\end{align*}

\textbf{Example 3.4.3}

\begin{align*}
    \int x e^x \, {\rm d}x
\end{align*}
Let $u = x$ so ${\rm d} u/{\rm d}x = 1$.
And ${\rm d} v/{\rm d}x = e^x$ so $v = e^x$.
Since $\int u\, {\rm d}v = uv - \int v \, {\rm d} u$ we get
\begin{align*}
    \int x e^x \, {\rm d}x &= x e^x - \int e^x \, {\rm d} x\\
    &= (x-1)e^x + C
\end{align*}

\textbf{Example 3.4.4}

\begin{align*}
    \int \tan^{-1}x \, {\rm d}x
\end{align*}
Let $u = \tan^{-1}x$ so ${\rm d} u/{\rm d}x = 1/(1+x^2)$.
And ${\rm d} v/{\rm d}x = 1$ so $v = x$.
Since $\int u\, {\rm d}v = uv - \int v \, {\rm d} u$ we get
\begin{align*}
    \int \tan^{-1}x \, {\rm d}x &= x \tan^{-1}x - \int \frac{x}{1+x^2} \, {\rm d} x\\
\end{align*}
Using the substitution $w = 1+x^2$, ${\rm d}w = 2x \, {\rm d}x$, therefore $x {\rm d}x = {\rm d}w / 2$, then
\begin{align*}
    \int \tan^{-1}x \, {\rm d}x &= x \tan^{-1}x - \int \frac{1}{2w} {\rm d}w\\
    &= x \tan^{-1}x - \frac{1}{2}\ln w + C\\
    &= x \tan^{-1}x - \frac{1}{2}\ln(1+x^2) + C
\end{align*}
