\textbf{Example 2.4.1}

\begin{align*}
    x^4 + 2xy + y^4 = 1 \\
\end{align*}
Differentiating w.r.t. $x$
\begin{align*}
    4x^3 + 2y + 2x\frac{{\rm d}y}{{\rm d}x} + 4y^3\frac{{\rm d}y}{{\rm d}x} &= 1 \\
    (2x + 4y^3)\frac{{\rm d}y}{{\rm d}x} &= -4x^3 - 2y\\
    \frac{{\rm d}y}{{\rm d}x} &= -\frac{4x^3 + 2y}{2x + 4y^3}\\
    &= -\frac{2x^3 + y}{x + 2y^3}
\end{align*}
At $x=1$ and $y=0$, this gives ${\rm d}y/{\rm d}x = -2$. Differentiate the above using the quotient rule, $u=-(2x^3+y)$ and $v=x+2y^3$, so $u'=-6x^2-{\rm d}y/{\rm d}x$ and $v' = 1 + 6y^2 {\rm d}y/{\rm d}x$
\begin{align*}
    \frac{{\rm d}^2y}{{\rm d}x^2} &= \frac{-(6x^2+{\rm d}y/{\rm d}x)(x+y^3)+(2x^3+y)(1 + 6y^2 {\rm d}y/{\rm d}x)}{(x+2y^3)^2}\\
\end{align*}
Substituting in $x=1$ and $y=0$ and ${\rm d}y/{\rm d}x = -2$, we find 
\begin{align*}
    \frac{{\rm d}^2y}{{\rm d}x^2} &= \frac{-(6-2)\times 1 + 2 \times 1}{1^2}\\
    &= -2 \,.
\end{align*}

\textbf{Example 2.4.2}

\begin{align*}
    y = \frac{\sqrt{x+a}\sqrt{x+b}}{x-c}
\end{align*}
Taking the log of both sides
\begin{align*}
    \ln y = \frac{1}{2} \ln(x+a) + \frac{1}{2} \ln(x+b) -  \ln(x-c)
\end{align*}
Taking the derivative
\begin{align*}
    \frac{1}{y}\frac{{\rm d}y}{{\rm d}x} = \frac{1}{2(x+a)} +\frac{1}{2(x+b)} -  \frac{1}{x-c}
\end{align*}
Multiplying through by $y$ and substituting in $y = \sqrt{x+a}\sqrt{x+b}/(x-c)$
\begin{align*}
    \frac{{\rm d}y}{{\rm d}x} = \frac{\sqrt{x+a}\sqrt{x+b}}{x-c}\left[\frac{1}{2(x+a)} +\frac{1}{2(x+b)} -  \frac{1}{x-c}\right]
\end{align*}

\textbf{Example 2.4.3}

\begin{align*}
x=t^3+t \qquad y=t^5+t^2
\end{align*}
So
\begin{align*}
\dot{x}=3t^2+1 \qquad \dot{y}=5t^4+2t
\end{align*}
Therefore
\begin{align*}
\frac{{\rm d}y}{{\rm d}x} = \frac{5t^4+2t}{3t^2+1} \equiv z
\end{align*}
Using the quotient rule
\begin{align*}
\dot{z} = \frac{(20t^3+2)(3t^2+1)-6t(5t^4+2t)}{(3t^2+1)^2}
\end{align*}
Therefore
\begin{align*}
\frac{{\rm d}^2y}{{\rm d}x^2} &= \frac{{\rm d}z}{{\rm d}x} = \frac{(20t^3+2)(3t^2+1)-6t(5t^4+2t)}{(3t^2+1)^3}\\
&= \frac{(20t^3+2)}{(3t^2+1)^2}-6t\frac{(5t^4+2t)}{(3t^2+1)^3}
\end{align*}
