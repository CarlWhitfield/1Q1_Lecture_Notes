\textbf{Example 1.2.1}

\begin{align*}
y &= 2x_0 - 2x_0 e^{-\alpha t}\\
y &= 2x_0 - 2x\\
y &= 2(x_0 - x)
\end{align*}
$t \geq 0$ so \par
$x = x_0$ at $t=0$ and $x=0$ as $t\rightarrow\infty$ \par
$y = 0$ at $t=0$ and $y \rightarrow 2 x_0$ as $t\rightarrow\infty$ \par
So $0 \leq x \leq x_0$ and $0 \leq y \leq 2x_0$.

\textbf{Example 1.3.1}

$$2x^2+5x-3 = (x+3)(2x-1)$$
\begin{align*}
\frac{x - 4}{(x+3)(2x-1)} &= \frac{A}{x+3} + \frac{B}{2x-1},\\
\therefore x-4 &= A(2x-1) + B(x+3)
\end{align*}
$$x=1/2 \Rightarrow 7B/2 =-7B/2 \Rightarrow B=-1 $$ \par
$$x=-3 \Rightarrow -7 = -7A \Rightarrow A=1$$ \par
so 
\begin{align*}
\frac{x - 4}{(x+3)(2x-1)} &= \frac{1}{x+3} - \frac{1}{2x-1}
\end{align*}
