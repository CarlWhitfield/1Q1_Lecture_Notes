\begin{align*}
z_1 + z_2 &= (5+2) + (3-7)i = 7 - 4i\\

\textbf{Example 5.2.2} \par

$z_1 = 3+4i$ and $z_2 = 5-5i$, so
\begin{align*}
z_1z_2 = (3+4i)(5-5i) = 15 + 20i - 15i - 20i^2 = (15+20) + (20-15)i = 35 + 5i
\end{align*}

\textbf{Example 5.2.3} \par

$z_1 = 3+4i$ and $z_2 = 5-5i$, so
\begin{align*}
\frac{z_1}{z_2} = \frac{3+4i}{5-5i} = \frac{(3+4i)(5+5i)}{(5-5i)(5+5i)} = \frac{15 + 20i + 15i + 20i^2}{5^2 + 5^2} = \frac{-5 +35i}{50} = -\frac{1}{10} + \frac{7}{10}i
\end{align*}

\textbf{Example 5.3.1} \par

$z = 5[\cos(\pi/3) + i \sin(\pi/3)]$, since $\cos(\pi/3)=1/2$ and $\sin(\pi/3) = \sqrt{3}/2$
\begin{align*}
    z = 5\left(\frac{1}{2} + \frac{\sqrt{3}}{2}i \right) = \frac{5}{2} + \frac{5\sqrt{3}}{2}i
\end{align*}

\textbf{Example 5.3.2} \par

$z = -2 + 2i$ so $|z| = \sqrt{2^2 + 2^2} = 2\sqrt{2}$. It is in the second quadrant so looking for an angle between $\pi/2$ and $\pi$. $\theta = \tan^{-1}(-1) = 3\pi/4$. Therefore
\begin{align*}
    z = 2\sqrt{2}\left[\cos\left(\frac{3\pi}{4}\right) + i \sin\left(\frac{3\pi}{4}\right)\right]
\end{align*}
