\begin{align*}
    h = k\left[4 + \left(\frac{x}{w}\right)^2 -4\left(\frac{x}{w} \right)^3\right]
\end{align*}

\begin{align*}
\int_0^w h \,{\rm d}x &= \left[kx\left(4 + \frac{x^2}{3w^2} - \frac{x^3}{w^3}\right)\right]_0^w \\
&= kw\left(4+\frac{1}{3}-1\right) \\
&= \frac{10}{3}kw
\end{align*}

\textbf{Example 3.2.2}

\begin{align*}
    f(x) &= 5x^2 - 13x + 13\\
    g(x) &= 2x + 3
\end{align*}
Find the crossing points
\begin{align*}
    5x^2 - 13x + 13 &= 2x + 3\\
    5x^2 - 15x + 10 &= 0\\
    x^2 - 3x + 2 &= 0\\
    (x-2)(x-1) = 0
\end{align*}
Therefore, crossing points at $x=1$ and $x=2$. Now integrate
\begin{align*}
    \int_1^2 [f(x) - g(x)] \,{\rm d} x &= \int_1^2 5(x^2 - 3x + 2) \,{\rm d} x \\
    &= 5\left[\frac{x^3}{3} - \frac{3x^2}{2} + 2x\right]_1^2\\
    &= 5\left[\frac{8}{3} - 6 + 4 - \frac{1}{3} + \frac{3}{2} - 2 \right] \\
    &= -\frac{5}{6}
\end{align*}
So area is 5/6.
